// 💼 BOLSA DE TRABAJO (CLICKEABLE) async function loadJobBoard() { const container = document.getElementById('job-board'); try { const { data, error } = await supabaseClient.from('job_board').select('*').order('created_at', { ascending: false }).limit(10); if (error) throw error; container.innerHTML = ''; if (!data || data.length === 0) { container.innerHTML = '

Sin ofertas activas.

'; return; } data.forEach(job => { const div = document.createElement('div'); div.className = 'job-card'; div.style.cssText = 'background:rgba(0,0,0,0.4); border-left:3px solid var(--neon-pink); padding:12px; margin:8px 0; border-radius:0 6px 6px 0; cursor:pointer; transition:0.3s;'; div.onclick = () => viewJobDetails(job); const typeIcon = job.type === 'barbero' ? '💈' : '🏪'; const typeText = job.type === 'barbero' ? 'Barbero busca trabajo' : 'Barbería busca personal'; div.innerHTML = `
${typeIcon} ${typeText} ${new Date(job.created_at).toLocaleDateString()}

${job.description}

Toca para contactar →

`; container.appendChild(div); }); } catch (err) { container.innerHTML = '

Error cargando ofertas.

'; console.error(err); } } // VER DETALLES DE OFERTA function viewJobDetails(job) { const contactMethod = confirm(`¿Contactar por WhatsApp?\n\n${job.description}\n\nCancelar para copiar número`) ? window.open(`https://wa.me/${job.contact}`, '_blank') : navigator.clipboard.writeText(job.contact).then(() => alert('Número copiado: ' + job.contact)); }
